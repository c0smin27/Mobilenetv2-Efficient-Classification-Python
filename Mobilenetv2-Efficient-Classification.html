Efficient Classification with Depthwise Separable Convolutions¶

Obiectiv¶

Clasificarea imaginilor din Fashion-MNIST (28x28, alb-negru) in 10 clase folosind arhitectura MobileNetV2, optimizata pentru eficienta pe dispozitive mobile si embedded.

De ce MobileNetV2¶

Modelul foloseste inverted residuals si depthwise separable convolutions, reducand semnificativ costul de calcul si numarul de parametri comparativ cu CNN-urile standard.
Este o alegere potrivita pentru modele usoare, cu viteza mare de inferenta si consum redus de memorie.

Metrice urmarite¶

  • Accuracy si F1 macro pe setul de test
  • Numar de parametri ai modelului
  • Timp mediu per epoca

Setari pentru modelul de baza (baseline)¶

  • Rezolutie de intrare: 96x96 pixeli
  • Ponderi initiale: preantrenate pe ImageNet
  • Augmentari usoare: aplicate la pasul 3
  • Divizare date: stratificata (train 54k / val 6k / test 10k)
  • Optimizator: Adam, LR initial = 1e-3

Scop general¶

Obtinerea unui model rapid si compact, cu un compromis echilibrat intre viteza si acuratete, demonstrand avantajele arhitecturii MobileNetV2 pentru clasificarea imaginilor Fashion-MNIST.

from tensorflow.keras.applications.mobilenet_v2 import preprocess_input

AUTOTUNE = tf.data.AUTOTUNE # optimizare paralelizare si prefetch
IMG_SIZE = 96               # rezolutie imagini
BATCH_SIZE = 128            # numarul de imagini / pas antrenare

# Conversie la RGB si redimensionare
def preprocess_no_scale(x, y):
    x = tf.expand_dims(x, -1)                    # (28, 28, 1)
    x = tf.image.grayscale_to_rgb(x)             # (28, 28, 3)
    x = tf.image.resize(x, (IMG_SIZE, IMG_SIZE)) # (96, 96, 3)
    return x, y

# Scalare pentru MobileNetV2
def scale_for_mobilenet(x, y):
    x = x * 255.0           # (0, 255)
    x = preprocess_input(x) # scaleaza la [-1, 1] pentru ImageNet
    return x, y

# Strat de augmentare (aplicat doar pe train)
data_augmentation = keras.Sequential([
    layers.RandomFlip("horizontal", seed=SEED),
    layers.RandomTranslation(0.05, 0.05, seed=SEED),
    layers.RandomRotation(0.05, seed=SEED),
    layers.RandomZoom(0.05, seed=SEED),
    layers.RandomContrast(0.05, seed=SEED),
], name="aug")

def make_ds(x, y, train=False):
    # Creare obiect pereche (imagine, eticheta)
    ds = tf.data.Dataset.from_tensor_slices((x, y))
    # Randomizare set de antrenare
    if train:
        ds = ds.shuffle(8192, seed=SEED)
    # Aplica functia de preprocesare pentru fiecare imagine
    ds = ds.map(preprocess_no_scale, num_parallel_calls=AUTOTUNE)
    # Aplica augmentari doar pe setul de antrenare
    if train:
      ds = ds.map(lambda xi, yi: (data_augmentation(xi, training=True), yi),
                  num_parallel_calls=AUTOTUNE)
    # Scalare imagini in [-1,1]
    ds = ds.map(scale_for_mobilenet, num_parallel_calls=AUTOTUNE)
    # Creare batch-uri si pregatire prefetch
    ds = ds.batch(BATCH_SIZE).prefetch(AUTOTUNE)
    return ds

train_ds = make_ds(x_train, y_train, train=True)
val_ds   = make_ds(x_val,   y_val)

# Model complet MobileNetV2 pentru clasificarea imaginilor Fashion-MNIST
def build_baseline_model(img_size=96, alpha=1.0, dropout=0.2, num_classes=10):
    # Tensor pentru imagini RGB (96×96×3)
    inputs = keras.Input(shape=(img_size, img_size, 3))

    # Stratul de augmentare
    aug = keras.Sequential([
        layers.RandomFlip("horizontal"),
        layers.RandomTranslation(0.05, 0.05),
        layers.RandomRotation(0.05),
        layers.RandomZoom(0.05),
        layers.RandomContrast(0.05),
    ], name="aug_layer")

    # Aplicare augmentari
    x = aug(inputs, training=True)

    # Backbone MobileNetV2
    base = keras.applications.MobileNetV2(
        input_shape=(img_size, img_size, 3),
        include_top=False,  # elimina ultimul strat de clasificare
        weights="imagenet", # invatate anterior
        alpha=alpha         # factor de latime (numarul de canale)
    )
    base.trainable = False  # ingheata toate straturile

    # Adaugarea capului de clasificare
    x = base(x)
    x = layers.GlobalAveragePooling2D()(x) # vector 1280 valori medii
    x = layers.Dropout(dropout)(x)         # dezactivare 20% unitati
    outputs = layers.Dense(num_classes, activation="softmax")(x)

    model = keras.Model(inputs, outputs, name=f"mnv2_a{alpha}_s{img_size}")
    return model

def plot_histories(h_list, labels):
    import numpy as np
    accs, val_accs = [], []
    losses, val_losses = [], []
    offset = 0 # continua numerotarea epocilor intre faze

    # Extragerea valorilor inregistrate de Keras la fiecare epoca
    for h, lab in zip(h_list, labels):
        a = h.history["accuracy"]      # acuratete antrenare
        va = h.history["val_accuracy"] # acuratete validare
        l = h.history["loss"]          # eroare la antrenare
        vl = h.history["val_loss"]     # eroare la validare
        e = np.arange(offset + 1, offset + len(a) + 1) # vector de epoci
        # Desenarea linillor
        plt.plot(e, a, label=f"{lab} acc")
        plt.plot(e, va, label=f"{lab} val_acc", linestyle="--")
        offset += len(a)

    plt.xlabel("Epoch")
    plt.ylabel("Accuracy")
    plt.title("Train vs Validation Accuracy")
    plt.legend()
    plt.grid(True)
    plt.show()

import time, os
import numpy as np
import pandas as pd
import tensorflow as tf
from tensorflow import keras
from tensorflow.keras import layers
from tensorflow.keras.applications.mobilenet_v2 import preprocess_input

# Configuratii test
exp_configs = [
    {"IMG_SIZE": 96,  "ALPHA": 1.0},
    {"IMG_SIZE": 128, "ALPHA": 1.0},
    {"IMG_SIZE": 128, "ALPHA": 0.75},
]

# Hiperparametri pentru CPU
WARMUP_EPOCHS = 2      # inghetat
FT_EPOCHS     = 2      # fine-tuning
LR_WARMUP     = 1e-3
LR_FT         = 1e-4
DROPOUT       = 0.2
BATCH_EXP     = 64
AUTOTUNE      = tf.data.AUTOTUNE

# Dataset pentru o rezolutie data
def make_ds_for_size(x, y, img_size, train=False):
    # Redimensionare, conversie RGB, scalare, batching
    def _pp(xi, yi):
        xi = tf.expand_dims(xi, -1)                 # (28,28,1)
        xi = tf.image.grayscale_to_rgb(xi)          # (28,28,3)
        xi = tf.image.resize(xi, (img_size, img_size))
        xi = preprocess_input(xi * 255.0)           # [-1,1] MobileNetV2
        return xi, yi
    ds = tf.data.Dataset.from_tensor_slices((x, y))
    if train:
        ds = ds.shuffle(8192, seed=SEED)
    ds = ds.map(_pp, num_parallel_calls=AUTOTUNE)
    ds = ds.batch(BATCH_EXP).prefetch(AUTOTUNE)
    return ds

# Augmentari doar in tf.data
aug_warm = keras.Sequential([
    layers.RandomFlip("horizontal", seed=SEED),
    layers.RandomTranslation(0.05, 0.05, seed=SEED),
    layers.RandomRotation(0.05, seed=SEED),
    layers.RandomZoom(0.05, seed=SEED),
    layers.RandomContrast(0.05, seed=SEED),
], name="aug_warm") # Augmentare warmup

aug_light = keras.Sequential([
    layers.RandomFlip("horizontal", seed=SEED),
    layers.RandomRotation(0.03, seed=SEED),
], name="aug_light") # Augmentare fine-tuning

# Model baseline MobileNetV2 fara augmentare interna
def build_baseline_model(img_size=96, alpha=1.0, dropout=0.2, num_classes=10):
    # Tensor pentru imagini RGB (96×96×3)
    inputs = keras.Input(shape=(img_size, img_size, 3))
    # Backbone MobileNetV2
    base = keras.applications.MobileNetV2(
        input_shape=(img_size, img_size, 3),
        include_top=False,  # elimina ultimul strat de clasificare
        weights="imagenet", # invatate anterior
        alpha=alpha         # factor de latime (numarul de canale)
    )
    base.trainable = False  # ingheata toate straturile
    x = base(inputs)
    x = layers.GlobalAveragePooling2D()(x) # vector 1280 valori medii
    x = layers.Dropout(dropout)(x)         # dezactivare 20% unitati
    outputs = layers.Dense(num_classes, activation="softmax")(x)
    return keras.Model(inputs, outputs, name=f"mnv2_a{alpha}_s{img_size}")

def run_experiment(cfg):
    img_size, alpha = cfg["IMG_SIZE"], cfg["ALPHA"]

    # Date la rezolutia ceruta
    tr_ds = make_ds_for_size(x_train, y_train, img_size, train=True)
    va_ds = make_ds_for_size(x_val,   y_val,   img_size, train=False)
    te_ds = make_ds_for_size(x_test,  y_test,  img_size, train=False)

    # Augmentare in tf.data doar pe train
    tr_warm  = tr_ds.map(lambda x,y: (aug_warm(x,  training=True), y),
                         num_parallel_calls=AUTOTUNE).prefetch(AUTOTUNE)
    tr_light = tr_ds.map(lambda x,y: (aug_light(x, training=True), y),
                         num_parallel_calls=AUTOTUNE).prefetch(AUTOTUNE)

    # Model baseline, backbone inghetat
    model = build_baseline_model(img_size=img_size, alpha=alpha, dropout=DROPOUT, num_classes=10)

    # Callbacks per-config, salveaza cel mai bun model pe val_accuracy
    ckpt_cfg = os.path.join(OUT_DIR, f"mnv2_a{alpha}_s{img_size}.keras")
    cbs = [
        keras.callbacks.EarlyStopping(monitor="val_accuracy", patience=2, restore_best_weights=True),
        keras.callbacks.ReduceLROnPlateau(monitor="val_loss", factor=0.5, patience=1, min_lr=1e-5),
        keras.callbacks.ModelCheckpoint(ckpt_cfg, monitor="val_accuracy", save_best_only=True),
    ]

    # Faza 1 - freeze: doar capul se antreneaza, LR mai mare
    model.compile(optimizer=keras.optimizers.Adam(LR_WARMUP),
                  loss="sparse_categorical_crossentropy", metrics=["accuracy"])
    t0 = time.time()
    h1 = model.fit(tr_warm, validation_data=va_ds, epochs=WARMUP_EPOCHS, callbacks=cbs, verbose=1)

    # Faza 2 - fine-tune: backbone deblocat, BatchNorm ramane inghetat, LR mic
    base = None
    for layer in model.layers:
        if "mobilenetv2" in layer.name.lower():
            base = layer; break
    base.trainable = True
    for l in base.layers:
        if isinstance(l, keras.layers.BatchNormalization):
            l.trainable = False

    model.compile(optimizer=keras.optimizers.Adam(LR_FT),
                  loss="sparse_categorical_crossentropy", metrics=["accuracy"])
    h2 = model.fit(tr_light, validation_data=va_ds, epochs=FT_EPOCHS, callbacks=cbs, verbose=1)
    t1 = time.time()

    # Evaluare pe test
    test_loss, test_acc = model.evaluate(te_ds, verbose=0)
    # Timpi si epoci
    n_ep = len(h1.history["accuracy"]) + len(h2.history["accuracy"])
    sec_per_epoch = (t1 - t0) / max(n_ep, 1)

    return {
        "IMG_SIZE": img_size,
        "ALPHA": alpha,
        "params": model.count_params(),
        "best_val_acc": float(max(h1.history["val_accuracy"] + h2.history["val_accuracy"])),
        "test_acc": float(test_acc),
        "sec_per_epoch": float(sec_per_epoch),
        "epochs_run": n_ep,
    }

# Ruleaza configurarile
results = []
for cfg in exp_configs:
    print(f"\n--- Run ---")
    print(f"IMG_SIZE = {cfg['IMG_SIZE']} | ALPHA = {cfg['ALPHA']}")
    res = run_experiment(cfg)
    results.append(res)
    print(res)

# Tabel sumar
df = pd.DataFrame(results).sort_values(by=["test_acc","params"], ascending=[False, True])
print("\nSummary:")
print(df.to_string(index=False))

def show_predictions_grid(model, x_test, y_test, classes, img_size=IMG_SIZE, n=10, n_rows=2, n_cols=5):
    from tensorflow.keras.applications.mobilenet_v2 import preprocess_input

    idxs = np.random.choice(len(x_test), n, replace=False)
    plt.figure(figsize=(n_cols * 2.5, n_rows * 2.5))

    for i, idxx in enumerate(idxs):
        im = x_test[idxx]
        true_lab = classes[int(y_test[idxx])]

        # Preprocesare identica cu antrenarea
        im_rgb = tf.image.grayscale_to_rgb(tf.expand_dims(im, -1))
        im_res = tf.image.resize(im_rgb, (img_size, img_size))
        im_ready = preprocess_input(im_res * 255.0)

        p = model.predict(tf.expand_dims(im_ready, 0), verbose=0)[0]
        pred_lab = classes[int(np.argmax(p))]

        plt.subplot(n_rows, n_cols, i + 1)
        plt.imshow(im, cmap="gray")
        plt.axis("off")
        ok = (pred_lab == true_lab)
        plt.title(f"Pred: {pred_lab}\nTrue: {true_lab}", fontsize=9, color=("green" if ok else "red"))

    plt.tight_layout()
    plt.show()

# Rulare
show_predictions_grid(model, x_test, y_test, classes)

8. Rezultate si concluzii¶

  • Arhitectura: MobileNetV2 preantrenat, cu cap de clasificare nou si depthwise separable convolutions

  • Antrenare in 2 faze: warmup cu backbone inghetat, apoi fine-tuning cu BatchNorm inghetat si rata de invatare mica (LR)

  • Acuratete finala pe test:

    • ~0.875 pentru modelul de baza
    • ~0.919 pentru configuratia optima din experimente

  • Experimente (IMG_SIZE x α):

    • 96 x 96, α=1.0, 2.27M param, test_acc ≈ 0.919, timp pe epoca scazut
    • 128 x 128, α=1.0, 2.27M param, test_acc ≈ 0.915, timp pe epoca mai mare
    • 128 x 128, α=0.75, 1.39M param, test_acc ≈ 0.913, parametri redusi semnificativ

  • Concluzie: configuratia 96 x 96, α=1.0 ofera cel mai bun raport intre viteza, dimensiune si precizie pe datasetul Fashion-MNIST

  • Analiza erorilor: confuzii frecvente intre:

    • Shirt cu T-shirt si Coat
    • Sandal cu Sneaker si Ankle boot
